#include <windows.h>
#include <winuser.h>
#include <tchar.h>
HINSTANCE g_hInst;
LPCTSTR lpszClass = L"Window Class Name";
LPCTSTR lpszWindowName = L"Window Programming 2-4";
LRESULT CALLBACK WndProc(HWND hWnd, UINT iMessage, WPARAM wParam, LPARAM lParam);

int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPSTR lpszCmdParam, int nCmdShow)
{
	HWND hWnd;
	MSG Message;
	WNDCLASSEX WndClass;
	g_hInst = hInstance;
	WndClass.cbSize = sizeof(WndClass);
	WndClass.style = CS_HREDRAW | CS_VREDRAW;
	WndClass.lpfnWndProc = (WNDPROC)WndProc;
	WndClass.cbClsExtra = 0;
	WndClass.cbWndExtra = 0;
	WndClass.hInstance = hInstance;
	WndClass.hIcon = LoadIcon(NULL, IDI_APPLICATION);
	WndClass.hCursor = LoadCursor(NULL, IDC_ARROW);
	WndClass.hbrBackground = (HBRUSH)GetStockObject(WHITE_BRUSH);
	WndClass.lpszMenuName = NULL;
	WndClass.lpszClassName = lpszClass;
	WndClass.hIconSm = LoadIcon(NULL, IDI_APPLICATION);
	RegisterClassEx(&WndClass);
	hWnd = CreateWindow(lpszClass, lpszWindowName, WS_OVERLAPPEDWINDOW, 0, 0, 800, 600, NULL, (HMENU)NULL, hInstance, NULL);
	ShowWindow(hWnd, nCmdShow);
	UpdateWindow(hWnd);
	while (GetMessage(&Message, 0, 0, 0)) {
		TranslateMessage(&Message);
		DispatchMessage(&Message);
	}
	return Message.wParam;
}
LRESULT CALLBACK WndProc(HWND hWnd, UINT uMsg, WPARAM wParam, LPARAM lParam)
{
	PAINTSTRUCT ps;
	HDC hDC;

	struct Library {
		int Shape;
		RECT rect;
		int Thick;
	};

	static Library pictures[10];

	static TCHAR str[6][5];
	static int count = 0;
	static int PicCount = 0;
	static SIZE size;
	static int cheak = 0;
	static int All = 0;
	static int prior, next;
	
	HPEN hPen, oldPen;
	HBRUSH hBrush, oldBrush;
	static COLORREF PenColor, BrushColor;

	switch (uMsg) {
	case WM_CREATE:
		CreateCaret(hWnd, NULL, 5, 15);
		ShowCaret(hWnd);
		count = 0;
		PenColor = RGB(rand() % 255, rand() % 255, rand() % 255);
		BrushColor = RGB(rand() % 255, rand() % 255, rand() % 255);
		break;
	case WM_CHAR:
		hDC = GetDC(hWnd);

		if (wParam == VK_BACK && cheak < 6) {
			if (str[cheak - 1][count - 1] == ' ') {
				cheak--;
			}
			str[cheak][--count] == '\0';
		}
		else if (wParam == VK_RETURN);
		else if (wParam == VK_SPACE && cheak < 6) {
			str[cheak][count++] = wParam;
			str[cheak][count] = '\0';
			cheak++;
			count = 0;
		}
		else if (wParam == '+') {
			if (cheak == 6) {
				if (pictures[PicCount].Thick == 10) {
					if (pictures[PicCount].rect.left != 0 || pictures[PicCount].rect.right != 800 || pictures[PicCount].rect.top != 0 || pictures[PicCount].rect.bottom != 600) {
						pictures[PicCount].rect.left += 10;
						pictures[PicCount].rect.top += 10;
						pictures[PicCount].rect.right += 10;
						pictures[PicCount].rect.bottom += 10;
					}
				}
				else
					pictures[PicCount].Thick += 1;
			}
			else {
				static RECT messageRect = { 0,0,800,600 };
				DrawText(hDC, _T("Error"), _tcslen(_T("Error")), &messageRect, DT_SINGLELINE | DT_CENTER | DT_VCENTER);
			}
		}
		else if (wParam == '-') {
			if (cheak == 6) {
				if (pictures[PicCount].Thick == 1) {
					pictures[PicCount].rect.left -= 10;
					pictures[PicCount].rect.top -= 10;
					pictures[PicCount].rect.right -= 10;
					pictures[PicCount].rect.bottom -= 10;
					if (pictures[PicCount].rect.left == pictures[PicCount].rect.right || pictures[PicCount].rect.top == pictures[PicCount].rect.bottom) {
						pictures[PicCount].rect.left += 10;
						pictures[PicCount].rect.top += 10;
						pictures[PicCount].rect.right += 10;
						pictures[PicCount].rect.bottom += 10;
					}
				}
				else
					pictures[PicCount].Thick -= 1;
			}
			else {
				static RECT messageRect = { 0,0,800,600 };
				DrawText(hDC, _T("Error"), _tcslen(_T("Error")), &messageRect, DT_SINGLELINE | DT_CENTER | DT_VCENTER);
			}
		}
		else if (wParam == '1'&&cheak==6)
				PenColor = RGB(rand() % 255, rand() % 255, rand() % 255);
		else if (wParam == '2'&&cheak==6) 
				BrushColor = RGB(rand() % 255, rand() % 255, rand() % 255);
		else if (wParam == 'p') {
			if (cheak == 6) {
				prior = 1;
			}
			else {
				static RECT messageRect = { 0,0,800,600 };
				DrawText(hDC, _T("Error"), _tcslen(_T("Error")), &messageRect, DT_SINGLELINE | DT_CENTER | DT_VCENTER);
			}
		}
		else if (wParam == 'n') {
			if (cheak == 6) {
				if (prior == 1)
					next = 1;
			}
			else {
				static RECT messageRect = { 0,0,800,600 };
				DrawText(hDC, _T("Error"), _tcslen(_T("Error")), &messageRect, DT_SINGLELINE | DT_CENTER | DT_VCENTER);
			}
		}
		else if (wParam == 'a') {
			if (cheak == 6) {
				All = 1;
			}
			else {
				static RECT messageRect = { 0,0,800,600 };
				DrawText(hDC, _T("Error"), _tcslen(_T("Error")), &messageRect, DT_SINGLELINE | DT_CENTER | DT_VCENTER);
			}
		}
		else if (wParam == 'q') {
			PostQuitMessage(0);
			break;
		}
		else if (cheak < 6) {
			if (0 <= wParam <= 9) {
				str[cheak][count++] = wParam;
				str[cheak][count] = '\0';
			}
			else {
				static RECT messageRect = { 0,0,800,600 };
				DrawText(hDC, _T("Error"), _tcslen(_T("Error")), &messageRect, DT_SINGLELINE | DT_CENTER | DT_VCENTER);
			}
		}
		else;

		InvalidateRect(hWnd, NULL, true);
		ReleaseDC(hWnd, hDC);
		break;
	case WM_PAINT:
		hDC = BeginPaint(hWnd, &ps);

		if (cheak == 6) {
			if (All == 1) {
				for (int i = 0; i <= PicCount; ++i) {
					hPen = CreatePen(PS_SOLID, pictures[i].Thick, PenColor); // 새로운 객체 만들기: 펜
					oldPen = (HPEN)SelectObject(hDC, hPen);
					hBrush = CreateSolidBrush(BrushColor); // 새로운 객체 만들기: 브러쉬
					oldBrush = (HBRUSH)SelectObject(hDC, hBrush);
					if (pictures[i].Shape == 1)
						SetPixel(hDC, pictures[i].rect.left, pictures[i].rect.top, PenColor);
					else if (pictures[i].Shape == 2) {
						MoveToEx(hDC, pictures[i].rect.left, pictures[i].rect.top, NULL);
						LineTo(hDC, pictures[i].rect.right, pictures[i].rect.bottom);
					}
					else if (pictures[i].Shape == 3) {
						POINT Triangle[3] = {
						{ (pictures[i].rect.left + pictures[i].rect.right) / 2, pictures[i].rect.top },
						{pictures[i].rect.left, pictures[i].rect.bottom},
						{pictures[i].rect.right, pictures[i].rect.bottom}
						};
						Polygon(hDC, Triangle, 3);
					}
					else if (pictures[i].Shape == 4)
						Rectangle(hDC, pictures[i].rect.left, pictures[i].rect.top, pictures[i].rect.right, pictures[i].rect.bottom);
					else if (pictures[i].Shape == 5) {
						POINT Pentagon[5] = {
						{(pictures[i].rect.left + pictures[i].rect.right) / 2, pictures[i].rect.top},
						{pictures[i].rect.left, (pictures[i].rect.top + pictures[i].rect.bottom) / 2},
						{pictures[i].rect.left + (pictures[i].rect.right - pictures[i].rect.left) / 6, pictures[i].rect.bottom},
						{pictures[i].rect.right - (pictures[i].rect.right - pictures[i].rect.left) / 6, pictures[i].rect.bottom},
						{pictures[i].rect.right,(pictures[i].rect.top + pictures[i].rect.bottom) / 2}
						};
						Polygon(hDC, Pentagon, 5);
					}
					else if (pictures[i].Shape == 6)
						Ellipse(hDC, pictures[i].rect.left, pictures[i].rect.top, pictures[i].rect.right, pictures[i].rect.bottom);
					SelectObject(hDC, oldPen);
					DeleteObject(hPen);
					SelectObject(hDC, oldBrush);
					DeleteObject(hBrush);
				}
			}
			else if (prior == 1) {

			}
			else if (next == 1) {

			}
			else {
				pictures[PicCount].Shape = _ttoi(str[0]);
				pictures[PicCount].rect.left = _ttoi(str[1]);
				pictures[PicCount].rect.top = _ttoi(str[2]);
				pictures[PicCount].rect.right = _ttoi(str[3]);
				pictures[PicCount].rect.bottom = _ttoi(str[4]);
				pictures[PicCount].Thick = _ttoi(str[5]);

				if (0 < pictures[PicCount].Shape <= 6 &&
					0 <= pictures[PicCount].rect.left <= 800 &&
					0 <= pictures[PicCount].rect.top <= 600 &&
					0 <= pictures[PicCount].rect.right <= 800 &&
					0 <= pictures[PicCount].rect.bottom <= 600 &&
					0 < pictures[PicCount].Thick < 11);
				else {
					CreateCaret(hWnd, NULL, 5, 15);
					ShowCaret(hWnd);
					cheak = 0;
					count = 0;
					for (int j = 0; j < 6; ++j) {
						for (int i = 0; i < _tcslen(str[j]); ++i)
							str[j][i] = '\0';
					}
					break;
				}

				hPen = CreatePen(PS_SOLID, pictures[PicCount].Thick, PenColor); // 새로운 객체 만들기: 펜
				oldPen = (HPEN)SelectObject(hDC, hPen);
				hBrush = CreateSolidBrush(BrushColor); // 새로운 객체 만들기: 브러쉬
				oldBrush = (HBRUSH)SelectObject(hDC, hBrush);

				{
					if (pictures[PicCount].Shape == 1)
						SetPixel(hDC, pictures[PicCount].rect.left, pictures[PicCount].rect.top, RGB(rand() % 255, rand() % 255, rand() % 255));
					else if (pictures[PicCount].Shape == 2) {
						MoveToEx(hDC, pictures[PicCount].rect.left, pictures[PicCount].rect.top, NULL);
						LineTo(hDC, pictures[PicCount].rect.right, pictures[PicCount].rect.bottom);
					}
					else if (pictures[PicCount].Shape == 3) {
						POINT Triangle[3] = {
						{ (pictures[PicCount].rect.left + pictures[PicCount].rect.right) / 2, pictures[PicCount].rect.top },
						{pictures[PicCount].rect.left, pictures[PicCount].rect.bottom},
						{pictures[PicCount].rect.right, pictures[PicCount].rect.bottom}
						};
						Polygon(hDC, Triangle, 3);
					}
					else if (pictures[PicCount].Shape == 4)
						Rectangle(hDC, pictures[PicCount].rect.left, pictures[PicCount].rect.top, pictures[PicCount].rect.right, pictures[PicCount].rect.bottom);
					else if (pictures[PicCount].Shape == 5) {
						POINT Pentagon[5] = {
						{(pictures[PicCount].rect.left + pictures[PicCount].rect.right) / 2, pictures[PicCount].rect.top},
						{pictures[PicCount].rect.left, (pictures[PicCount].rect.top + pictures[PicCount].rect.bottom) / 2},
						{pictures[PicCount].rect.left + (pictures[PicCount].rect.right - pictures[PicCount].rect.left) / 6, pictures[PicCount].rect.bottom},
						{pictures[PicCount].rect.right - (pictures[PicCount].rect.right - pictures[PicCount].rect.left) / 6, pictures[PicCount].rect.bottom},
						{pictures[PicCount].rect.right,(pictures[PicCount].rect.top + pictures[PicCount].rect.bottom) / 2}
						};
						Polygon(hDC, Pentagon, 5);
					}
					else if (pictures[PicCount].Shape == 6)
						Ellipse(hDC, pictures[PicCount].rect.left, pictures[PicCount].rect.top, pictures[PicCount].rect.right, pictures[PicCount].rect.bottom);
				}

				
				DestroyCaret();

				SelectObject(hDC, oldPen);
				DeleteObject(hPen);
				SelectObject(hDC, oldBrush);
				DeleteObject(hBrush);

				CreateCaret(hWnd, NULL, 5, 15);
				ShowCaret(hWnd);
				cheak = 0;
				count = 0;
				PicCount++;
				for (int j = 0; j < 6; ++j) {
					for (int i = 0; i < _tcslen(str[j]); ++i)
						str[j][i] = '\0';
				}
			}
		}
		else {
			GetTextExtentPoint32(hDC, str[cheak], lstrlen(str[cheak]), &size);
			for (int i = 0; i <= cheak; ++i) {
				TextOut(hDC, i*40, 0, str[i], _tcslen(str[i]));
			}
			SetCaretPos(size.cx + cheak * 40, 20);
		}

		EndPaint(hWnd, &ps);
		break;
	case WM_DESTROY:
		HideCaret(hWnd);
		PostQuitMessage(0);
		break;
		}
		return DefWindowProc(hWnd, uMsg, wParam, lParam);
	}
